// The newest 4:00 (server time), as a gettimetick(2): a daily quest done before it can be done again.
function	script	F_Bst_Reset	{
	return gettimetick(2) - ((gettime(DT_HOUR) + 20) % 24) * 3600 - gettime(DT_MINUTE) * 60 - gettime(DT_SECOND);
}

// Counts the kills of the active hunting quest.
-	script	Booster#hunt	-1,{
	end;

OnNPCKillEvent:
	if (!BST_HQ || !callfunc("F_Bst_Hunt", BST_HQ))
		end;
	.@mob = killedrid;
	.@lv = getmonsterinfo(.@mob, MOB_LV);
	.@map$ = strcharinfo(3);
	.@n = getarraysize(@bst_cnt);
	.@done = 0;
	for (.@t = 0; .@t < .@n; .@t++) {
		if (BST_HK[.@t] >= @bst_cnt[.@t]) {
			.@done++;
			continue;
		}
		explode(.@p$, @bst_tgt$[.@t], ":");
		.@hit = 0;
		if (.@p$[0] == "mob" || .@p$[0] == "map") {
			deletearray .@ids$;
			explode(.@ids$, .@p$[1], ",");
			for (.@k = 0; .@k < getarraysize(.@ids$); .@k++) {
				if ((.@p$[0] == "mob" && atoi(.@ids$[.@k]) == .@mob) || (.@p$[0] == "map" && .@ids$[.@k] == .@map$)) {
					.@hit = 1;
					break;
				}
			}
		} else if (.@lv >= atoi(.@p$[2]) && .@lv <= atoi(.@p$[3])) {
			if (.@p$[0] == "ele")
				.@hit = getmonsterinfo(.@mob, MOB_ELEMENT) == atoi(.@p$[1]);
			else if (.@p$[0] == "size")
				.@hit = getmonsterinfo(.@mob, MOB_SIZE) == atoi(.@p$[1]);
			else if (.@p$[0] == "race")
				.@hit = getmonsterinfo(.@mob, MOB_RACE) == atoi(.@p$[1]);
		}
		if (!.@hit)
			continue;
		BST_HK[.@t]++;
		dispbottom "[Booster] " + @bst_lbl$[.@t] + ": " + BST_HK[.@t] + "/" + @bst_cnt[.@t];
		if (BST_HK[.@t] >= @bst_cnt[.@t])
			.@done++;
		.@counted = 1;
	}
	if (.@counted && .@done == .@n)
		dispbottom "[Booster] Hunting quest done! Ring your Booster Call to report to the Booster Assistant.";
	end;
}

// Summoned next to the player by the Booster Call (item 100337) with duplicate_dynamic.
-	script	Booster Assistant#bst	4_F_EDEN_OFFICER,{
	function Bst_Hunting;
	function Bst_HuntInfo;
	function Bst_Collection;
	function Bst_Help;

	.@n$ = "[Booster Assistant]";
	mes .@n$;
	mes "Hello, " + strcharinfo(0) + "!";
	mes "How can I help your adventure today?";
	next;
	while (true) {
		switch (select("Booster Coin:Hunting Quests:Item Collection Quests:Additional Help:Cancel")) {
		case 1:
			switch (select("Booster Coin Shop:Exchange EXP potions for Booster Coins:Back")) {
			case 1:
				close2;
				callshop "booster_coin_shop";
				end;
			case 2:
				mes .@n$;
				mes "Five Growth Potions of a kind, or fifteen Job Potions, make one Booster Coin.";
				close2;
				callshop "booster_coin_exchange";
				end;
			}
			break;
		case 2:
			Bst_Hunting();
			break;
		case 3:
			Bst_Collection();
			break;
		case 4:
			Bst_Help();
			break;
		default:
			mes .@n$;
			mes "Ring the Booster Call whenever you need me!";
			close;
		}
	}
	end;

	// The active quest, to report or give up; else the quests to take.
	function Bst_Hunting {
		.@n$ = "[Booster Assistant]";
		if (BST_HQ) {
			callfunc "F_Bst_Hunt", BST_HQ;
			Bst_HuntInfo(BST_HQ, 1);
			.@done = 1;
			for (.@t = 0; .@t < getarraysize(@bst_cnt); .@t++)
				if (BST_HK[.@t] < @bst_cnt[.@t])
					.@done = 0;
			if (.@done) {
				if (select("Report the quest:Later") == 2)
					return;
				if (!checkweight(1000254, @bst_coin, 100346, max(1, @bst_job)) || (@bst_pot && !checkweight(@bst_pot, @bst_potn))) {
					mes .@n$;
					mes "You are carrying too much! Make some room in your bag first.";
					next;
					return;
				}
				.@i = BST_HQ;
				BST_HDONE[.@i] = gettimetick(2);
				BST_HQ = 0;
				deletearray BST_HK;
				getitem 1000254, @bst_coin;
				if (@bst_pot)
					getitem @bst_pot, @bst_potn;
				if (@bst_job)
					getitem 100346, @bst_job;
				mes .@n$;
				mes "Well done! Here is your reward.";
				mes "This quest can be done again after 4:00.";
				next;
				return;
			}
			if (select("Keep hunting:Give up this quest") == 2) {
				BST_HQ = 0;
				deletearray BST_HK;
				mes .@n$;
				mes "Quest given up. You can take another one now.";
				next;
			}
			return;
		}
		switch (select("Quests below Lv 99:Quests from Lv 99:Quest information:Back")) {
		case 1:
			.@from = 1;
			.@to = 9;
			break;
		case 2:
			.@from = 10;
			.@to = 20;
			break;
		case 3:
			mes .@n$;
			mes "Take one hunting quest at a time, at or above its level.";
			mes "Hunt anywhere: the kills count wherever you find the monsters, but a quest on a map counts only that map's monsters.";
			next;
			mes .@n$;
			mes "When you are done, ring the Booster Call and report to me for Booster Coins and potions.";
			mes "Each quest, and each item collection request, can be done once a day. They reset at 4:00.";
			next;
			return;
		default:
			return;
		}
		.@reset = callfunc("F_Bst_Reset");
		.@menu$ = "";
		for (.@i = .@from; .@i <= .@to; .@i++) {
			callfunc "F_Bst_Hunt", .@i;
			.@s$ = "[Lv " + @bst_lv + "] " + @bst_area$;
			if (BST_HDONE[.@i] >= .@reset)
				.@s$ += " (done today)";
			else if (BaseLevel < @bst_lv)
				.@s$ += " (too low)";
			.@menu$ += .@s$ + ":";
		}
		.@i = select(.@menu$ + "Back") - 1 + .@from;
		if (.@i > .@to)
			return;
		callfunc "F_Bst_Hunt", .@i;
		Bst_HuntInfo(.@i, 0);
		if (BST_HDONE[.@i] >= .@reset) {
			mes .@n$;
			mes "You have done this quest today. It resets at 4:00.";
			next;
			return;
		}
		if (BaseLevel < @bst_lv) {
			mes .@n$;
			mes "This quest needs Base Level " + @bst_lv + ".";
			next;
			return;
		}
		if (select("Take this quest:Back") == 2)
			return;
		BST_HQ = .@i;
		deletearray BST_HK;
		mes .@n$;
		mes "Quest taken. Good hunting!";
		mes "Your progress shows in the chat as you hunt.";
		next;
		return;
	}

	// Hunting quest getarg(0) (loaded by F_Bst_Hunt) with its targets; getarg(1): show the kills so far.
	function Bst_HuntInfo {
		mes "[Lv " + @bst_lv + "] " + @bst_area$;
		for (.@t = 0; .@t < getarraysize(@bst_cnt); .@t++) {
			if (getarg(1))
				mes "- " + @bst_lbl$[.@t] + ": " + min(BST_HK[.@t], @bst_cnt[.@t]) + "/" + @bst_cnt[.@t];
			else
				mes "- " + @bst_cnt[.@t] + "x " + @bst_lbl$[.@t];
		}
		mes "Reward: " + callfunc("F_Bst_Rewards");
		next;
		return;
	}

	// The item collection requests: hand the items over for coins, each once a day.
	function Bst_Collection {
		.@n$ = "[Booster Assistant]";
		.@reset = callfunc("F_Bst_Reset");
		.@menu$ = "";
		for (.@i = 1; .@i <= 11; .@i++) {
			callfunc "F_Bst_Collect", .@i;
			.@menu$ += "Request " + .@i + " (" + @bst_coin + " coin" + (@bst_coin > 1 ? "s" : "") + ")" + (BST_CDONE[.@i] >= .@reset ? " (done today)" : "") + ":";
		}
		.@i = select(.@menu$ + "Back");
		if (.@i > 11)
			return;
		explode(.@pair$, callfunc("F_Bst_Collect", .@i), ",");
		.@n = getarraysize(.@pair$);
		.@ok = 1;
		mes .@n$;
		mes "Request " + .@i + " needs:";
		for (.@k = 0; .@k < .@n; .@k++) {
			explode(.@p$, .@pair$[.@k], ":");
			.@id[.@k] = atoi(.@p$[0]);
			.@amount[.@k] = atoi(.@p$[1]);
			mes "- " + .@amount[.@k] + "x " + getitemname(.@id[.@k]) + " (" + countitem(.@id[.@k]) + ")";
			if (countitem(.@id[.@k]) < .@amount[.@k])
				.@ok = 0;
		}
		mes "Reward: " + @bst_coin + " " + getitemname(1000254);
		next;
		if (BST_CDONE[.@i] >= .@reset) {
			mes .@n$;
			mes "You have done this request today. It resets at 4:00.";
			next;
			return;
		}
		if (!.@ok) {
			mes .@n$;
			mes "Come back when you have all the items.";
			next;
			return;
		}
		if (select("Hand them over:Back") == 2)
			return;
		if (!checkweight(1000254, @bst_coin)) {
			mes .@n$;
			mes "You are carrying too much! Make some room in your bag first.";
			next;
			return;
		}
		for (.@k = 0; .@k < .@n; .@k++)
			delitem .@id[.@k], .@amount[.@k];
		BST_CDONE[.@i] = gettimetick(2);
		getitem 1000254, @bst_coin;
		mes .@n$;
		mes "Thank you! Here are your Booster Coins.";
		next;
		return;
	}

	function Bst_Help {
		.@n$ = "[Booster Assistant]";
		switch (select("How to play:Tips:Job Changer:Teleport to Prontera:Back")) {
		case 1:
			mes .@n$;
			mes "Open your Booster Pack: inside is another pack that opens at a higher level, and so on up to Lv 230. Each one has gear and supplies for that level.";
			next;
			mes .@n$;
			mes "Earn Booster Coins with my daily hunting quests and item collection requests, then spend them in the Booster Coin Shop.";
			mes "Equipment tickets go to the Booster Equipment NPC, in north Prontera by the castle.";
			next;
			break;
		case 2:
			mes .@n$;
			mes "- Hunting quests pay Growth Potions too: five of a kind (or fifteen Job Potions) trade for a Booster Coin.";
			mes "- A Metal Weapon Ticket costs only 1 coin: a solid weapon for any job.";
			mes "- Booster shadow gear costs 15 coins a piece.";
			mes "- Everything resets at 4:00, so check in daily!";
			next;
			break;
		case 3:
			if (getnpcid(0, "Job Master") == 0) {
				mes .@n$;
				mes "The Job Master is turned off in the app's NPC settings, so I can't call them.";
				next;
				break;
			}
			if (duplicate_dynamic("Job Master") == "") {
				mes .@n$;
				mes "The Job Master couldn't come here. Try again somewhere with more space.";
				next;
				break;
			}
			mes .@n$;
			mes "I've called the Job Master. They're right next to you!";
			close;
		case 4:
			mes .@n$;
			mes "Off to Prontera!";
			close2;
			warp "prontera", 167, 296;
			end;
		}
		return;
	}
}

		mes .@n$;
		mes "Come back any time!";
		close;
	}

		mes .@n$;
		mes "You are carrying too much! Make some room in your bag first.";
		close;
	}

		jobchange Job_Novice_High;
		resetlvl 1;
		specialeffect2 EF_ANGEL2;
	}

	getitem 100029, 1;
	mes .@n$;
	mes "You're registered! Use the " + getitemname(100337) + " any time.";
	if (.@high)
		mes "At Job Level 10, the Job Master makes you a transcended first job.";
